      private function NineTailsStrikePose(frame:TAnimationFrame) : void {
         if(FId!=11120994 || frame==null || stage==null || FTargets==null || FResults==null)return;
         var motion:Object=NineTailsVfxManifest().actorMotion[String(FSkillId)];if(motion==null)return;
         var enemy:TRole=null;
         for(var i:int=0;i<FTargets.length && i<FResults.length;i++)if(FTargets[i]!=null && FTargets[i].Camp!=Camp && FResults[i].CMD==1){enemy=FTargets[i];break;}
         if(enemy==null)return;
         var contact:Point=globalToLocal(enemy.localToGlobal(new Point()));
         var dx:Number=contact.x-(contact.x<0?-85:85);var phase:Number;
         if(FFrameIndex<=motion.first){phase=Math.max(0,Math.min(1,(FFrameIndex-motion.start)/Math.max(1,motion.first-motion.start)));}
         else if(FFrameIndex<=motion.last){phase=1;}
         else{phase=Math.max(0,Math.min(1,(motion.end-FFrameIndex)/Math.max(1,motion.end-motion.last)));}
         phase=phase*phase*(3-2*phase);
         FBitmap.x=-frame.Pivot.X+dx*phase;FBitmap.y=-frame.Pivot.Y+contact.y*phase;
      }
