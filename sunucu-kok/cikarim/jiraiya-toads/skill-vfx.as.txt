      private static function SetJiraiyaBarrier(host:DisplayObjectContainer, enabled:Boolean) : void
      {
         if(host == null) return;
         var existing:DisplayObject = host.getChildByName("jiraiyaEsophagusBackdrop");
         if(!enabled)
         {
            if(existing != null) host.removeChild(existing);
            return;
         }
         if(existing != null) return;
         var backdrop:Sprite = new Sprite();
         backdrop.name = "jiraiyaEsophagusBackdrop";
         backdrop.mouseEnabled = false;backdrop.mouseChildren = false;
         backdrop.graphics.beginFill(0x7A302D);backdrop.graphics.drawRect(0,0,1250,650);backdrop.graphics.endFill();
         host.addChildAt(backdrop,0);
         var loader:Loader = new Loader();
         var completed:Function;
         var failed:Function;
         var removed:Function;
         var release:Function = function():void
         {
            loader.contentLoaderInfo.removeEventListener(Event.COMPLETE,completed);
            loader.contentLoaderInfo.removeEventListener(IOErrorEvent.IO_ERROR,failed);
            loader.contentLoaderInfo.removeEventListener(SecurityErrorEvent.SECURITY_ERROR,failed);
         };
         completed = function(event:Event):void
         {
            release();
            if(backdrop.parent == null) return;
            loader.width=1250;loader.height=650;
            if(loader.content is Bitmap) Bitmap(loader.content).smoothing=true;
            backdrop.addChild(loader);
         };
         failed = function(event:Event):void {release();};
         removed = function(event:Event):void
         {
            if(event.target != backdrop) return;
            release();
            backdrop.removeEventListener(Event.REMOVED_FROM_STAGE,removed);
            try {loader.close();} catch(error:Error) {}
            loader.unload();
         };
         loader.contentLoaderInfo.addEventListener(Event.COMPLETE,completed);
         loader.contentLoaderInfo.addEventListener(IOErrorEvent.IO_ERROR,failed);
         loader.contentLoaderInfo.addEventListener(SecurityErrorEvent.SECURITY_ERROR,failed);
         backdrop.addEventListener(Event.REMOVED_FROM_STAGE,removed);
         loader.load(new URLRequest("/public/jiraiya-vfx/toad-esophagus-v1.jpg"));
      }

      private function JiraiyaRasenganPose() : Object
      {
         var hands:Array = __JIRAIYA_RASENGAN_HANDS__;
         var frame:int = Math.max(0,Math.min(48,FFrameIndex));
         var palm:Array = hands[frame];
         var contact:Point = new Point(90,-25);
         if(FTargets != null)
         {
            for(var i:int=0;i<FTargets.length;i++)
            {
               if(FTargets[i] != null && FTargets[i] != this && FTargets[i].Camp != Camp)
               {
                  contact = globalToLocal(FTargets[i].localToGlobal(new Point(0,-25)));
                  break;
               }
            }
         }
         var travel:Number = frame <= 22 ? 0 : (frame <= 30 ? Math.pow((frame-22)/8,1.4) : (frame <= 38 ? 1 : Math.max(0,1-(frame-38)/10)));
         var rise:Number = frame <= 4 ? 0 : (frame < 16 ? Math.sin((frame-4)/12*Math.PI/2) : (frame <= 22 ? 1 : (frame < 30 ? 1-Math.pow((frame-22)/8,2) : 0)));
         var ground:Point = FBattleStage.globalToLocal(localToGlobal(new Point(0,0)));
         // Keep the overhead sphere inside the battlefield even from a back-row slot.
         var lift:Number = Math.min(145,Math.max(0,ground.y+Number(palm[1])-160))*rise;
         return {x:(contact.x-Number(hands[30][0]))*travel,y:(contact.y-Number(hands[30][1]))*travel-lift,handX:Number(palm[0]),handY:Number(palm[1])};
      }

      private function UpdateJiraiyaVfx() : void
      {
         if(FId != 11120903 || FSkillId < 13120911 || FSkillId > 13120916 || (FIndex != TYPE_ACTIVE_ATTACK && FIndex != TYPE_ACTIVE_SKILL)) return;
         if(!FJiraiyaVfxStarted)
         {
            FJiraiyaVfxStarted = true;
            this.ShowJiraiyaVfx(FSkillId,false);
         }
         var hits:Object = __JIRAIYA_VFX_FRAMES__;
         if(!FJiraiyaVfxHit && FFrameIndex >= int(hits[FSkillId]))
         {
            FJiraiyaVfxHit = true;
            this.ShowJiraiyaVfx(FSkillId,true);
         }
      }

      private static function ShakeJiraiyaBattle(host:DisplayObjectContainer, strength:Number, duration:int) : void
      {
         if(host == null || host.stage == null) return;
         if(FJiraiyaShake != null) FJiraiyaShake.stop();
         var originX:Number = host.x;
         var originY:Number = host.y;
         var started:int = getTimer();
         var tick:Function;
         var stop:Function;
         var removed:Function;
         stop = function():void
         {
            host.removeEventListener(Event.ENTER_FRAME,tick);
            host.removeEventListener(Event.REMOVED_FROM_STAGE,removed);
            host.x = originX;host.y = originY;
            FJiraiyaShake = null;
         };
         removed = function(event:Event):void {if(event.target == host) stop();};
         tick = function(event:Event):void
         {
            var age:Number = getTimer() - started;
            if(age >= duration || host.stage == null) {stop();return;}
            var decay:Number = Math.pow(1-age/duration,2);
            host.x = originX + Math.sin(age*0.091)*strength*decay;
            host.y = originY + Math.cos(age*0.127)*strength*0.55*decay;
         };
         FJiraiyaShake = {stop:stop};
         host.addEventListener(Event.ENTER_FRAME,tick);
         host.addEventListener(Event.REMOVED_FROM_STAGE,removed);
      }

      private function ShowJiraiyaVfx(skill:int, impact:Boolean) : void
      {
         var host:DisplayObjectContainer = this.FBattleStage;
         if(host == null || host.stage == null) return;
         var caster:TRole = this;
         var origin:Point = host.globalToLocal(localToGlobal(new Point(0,-48)));
         var targets:Array = [];
         var seen:Object = {};
         var key:String;
         var n:int;
         if(FTargets != null)
         {
            for(n=0;n<FTargets.length;n++)
            {
               if(FTargets[n] == null || FTargets[n] == this || FResults == null || n >= FResults.length) continue;
               if(FResults[n].CMD != TTargetInfo.CMD_ATTACK && FResults[n].CMD != TTargetInfo.CMD_ATTACKEX) continue;
               if(FResults[n].ResultInfo.HurtHp < 0) continue;
               key = FTargets[n].Camp + "_" + FTargets[n].Pos;
               if(seen[key]) continue;
               seen[key] = true;
               targets.push(host.globalToLocal(FTargets[n].localToGlobal(new Point(0,-60))));
            }
         }
         if(targets.length == 0) targets.push(new Point(origin.x+(Camp == 0 ? 340 : -340),origin.y));
         var direction:Number = targets[0].x >= origin.x ? 1 : -1;
         var color:uint = skill == 13120913 ? 0x42DFFF : (skill == 13120912 ? 0xFFD45A : (skill == 13120911 ? 0xDE735C : (skill == 13120914 ? 0xE3F6FF : 0xFFE68A)));
         if(skill == 13120911 && impact) SetJiraiyaBarrier(host,true);
         var layer:Sprite = new Sprite();
         layer.name = "jiraiyaSkillVfx";
         layer.mouseEnabled = false;layer.mouseChildren = false;
         // Mouth curtains need opaque paint; chakra uses additive light.
         layer.blendMode = skill == 13120911 ? BlendMode.NORMAL : BlendMode.ADD;
         if(skill != 13120911) layer.filters = [new GlowFilter(color,0.8,12,12,1.6,1)];
         host.addChild(layer);
         var shade:Sprite = null;
         if(skill == 13120913 || skill == 13120912)
         {
            shade = new Sprite();shade.name="jiraiyaCinematicShade";
            shade.mouseEnabled=false;shade.mouseChildren=false;
            shade.graphics.beginFill(skill == 13120913 ? 0x041429 : 0x241406,0.65);
            shade.graphics.drawRect(0,0,1250,650);shade.graphics.endFill();
            host.addChildAt(shade,host.getChildByName("jiraiyaEsophagusBackdrop") == null ? 0 : 1);
         }
         var started:int = getTimer();
         var lifetime:int = skill == 13120913 ? 1050 : (skill == 13120912 ? 900 : (skill == 13120911 ? 560 : 720));
         var strength:Number = impact ? (skill == 13120913 ? 12 : (skill == 13120911 ? 9 : (skill == 13120912 ? 8 : 4))) : 1;
         ShakeJiraiyaBattle(host,strength,impact ? 440 : 180);
         var markers:Object = __JIRAIYA_VFX_FRAMES__;
         var animate:Function;
         var cleanup:Function;
         cleanup = function(event:Event = null):void
         {
            if(event != null && event.target != layer) return;

            layer.removeEventListener(Event.REMOVED_FROM_STAGE,cleanup);
            if(shade != null && shade.parent != null) shade.parent.removeChild(shade);
            if(layer.parent != null) layer.parent.removeChild(layer);
         };
         animate = function(event:Event):void
         {
            var elapsed:Number = getTimer()-started;
            var t:Number = impact ? elapsed/lifetime : Math.min(0.999,Math.max(0,caster.FFrameIndex)/Math.max(1,int(markers[skill])));
            if(host.stage == null || caster.stage == null || (impact && t >= 1) || (!impact && (caster.FSkillId != skill || caster.FJiraiyaVfxHit || elapsed > 8000)))
            {cleanup();return;}

            var g:Graphics = layer.graphics;g.clear();

            if(shade != null) shade.alpha=impact ? (1-t)*0.8 : Math.min(0.85,t*2);

            if(skill == 13120911)
            {
               // Upper/lower fleshy lips close the scene, then reveal the new interior.

               var lip:Number = 335*close;
               for(i=0;i<2;i++)

                  g.moveTo(0,i == 0 ? 0 : 650);
                  for(j=0;j<=20;j++)

                     g.lineTo(px,i == 0 ? py : 650-py);
                  }
                  g.lineTo(1250,i == 0 ? 0 : 650);g.endFill();
                  for(j=0;j<25;j++)

                     px=j*53;

                     g.moveTo(px,i == 0 ? 0 : 650);

                  }
               }
               if(impact && t < 0.26)

                  g.drawRect(0,0,1250,650);g.endFill();
               }

            }
            if(skill == 13120913)
            {

                  radius=20+growth*46;
                  var pose:Object = caster.JiraiyaRasenganPose();
