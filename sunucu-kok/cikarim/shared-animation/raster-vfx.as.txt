      private static function RasterTemplateVfxManifest() : Object
      {
         return JSON.parse(__RASTER_VFX_MANIFEST__);
      }

      private static function PreloadRasterTemplateVfx(heroId:int=0):void {
         var manifest:Object=RasterTemplateVfxManifest();
         for(var key:String in manifest.atlases)LoadRasterTemplateVfxAtlas(key,manifest.atlases[key]);
      }

      private static function LoadRasterTemplateVfxAtlas(name:String, meta:Object) : void
      {
         if(FRasterTemplateFxCache[name] != null || FRasterTemplateFxLoading[name]) return;
         FRasterTemplateFxLoading[name] = true;
         var loader:Loader = new Loader();
         var complete:Function;
         var failed:Function;
         var release:Function = function():void
         {
            loader.contentLoaderInfo.removeEventListener(Event.COMPLETE,complete);
            loader.contentLoaderInfo.removeEventListener(IOErrorEvent.IO_ERROR,failed);
            loader.contentLoaderInfo.removeEventListener(SecurityErrorEvent.SECURITY_ERROR,failed);
            delete FRasterTemplateFxLoading[name];
         };
         failed = function(event:Event):void {release();loader.unload();};
         complete = function(event:Event):void
         {
            release();
            var source:Bitmap = loader.content as Bitmap;
            if(source == null) {loader.unload();return;}
            var frames:Array = [];
            var w:int = int(source.bitmapData.width/int(meta.columns));
            var h:int = int(source.bitmapData.height/int(meta.rows));
            for(var i:int=0;i<(meta.frameCount != null ? int(meta.frameCount) : int(meta.columns)*int(meta.rows));i++)
            {
               var cell:BitmapData = new BitmapData(w,h,true,0);
               cell.copyPixels(source.bitmapData,new flash.geom.Rectangle((i%int(meta.columns))*w,int(i/int(meta.columns))*h,w,h),new Point());
               frames.push(cell);
            }
            FRasterTemplateFxCache[name] = frames;
            loader.unload();
         };
         loader.contentLoaderInfo.addEventListener(Event.COMPLETE,complete);
         loader.contentLoaderInfo.addEventListener(IOErrorEvent.IO_ERROR,failed);
         loader.contentLoaderInfo.addEventListener(SecurityErrorEvent.SECURITY_ERROR,failed);
         loader.load(new URLRequest(String(meta.url)));
      }

      private function PlayRasterTemplateVfxAsset(name:String, host:DisplayObjectContainer, point:Point, width:Number, height:Number, delay:int = 0, mirrored:Boolean = false, follow:Boolean = false, behind:Boolean = false) : void
      {
         var manifest:Object = RasterTemplateVfxManifest();
         var clip:Object = manifest.clips[name];
         if(clip == null || host == null || host.stage == null) return;
         LoadRasterTemplateVfxAtlas(String(clip.atlas),manifest.atlases[clip.atlas]);
         var caster:TRole = this;
         var layer:Sprite = new Sprite();
         layer.name = "rasterTemplateSkillVfx";layer.mouseEnabled = false;layer.mouseChildren = false;
         var image:Bitmap = new Bitmap();image.smoothing = true;layer.addChild(image);
         if(clip.softEdges)
         {
            // Keep broad translucent energy from exposing the rectangular cell boundary.
            var edgeMask:Sprite = new Sprite();
            var gradient:Matrix = new Matrix();
            gradient.createGradientBox(width,height,0,-width*Number(clip.pivot[0]),-height*Number(clip.pivot[1]));
            edgeMask.graphics.beginGradientFill(GradientType.RADIAL,[0xFFFFFF,0xFFFFFF],[1,0],[175,255],gradient);
            edgeMask.graphics.drawRect(-width*Number(clip.pivot[0]),-height*Number(clip.pivot[1]),width,height);
            edgeMask.graphics.endFill();
            image.cacheAsBitmap=true;edgeMask.cacheAsBitmap=true;layer.addChild(edgeMask);image.mask=edgeMask;
         }
         if(behind && host.parent != null)
         {
            // Character ZBufferSort rewrites every child index in the role container.
            // Put the eruption behind that container so depth sorting cannot cover roles.
            var field:DisplayObject = host;
            var outer:DisplayObjectContainer = host.parent;
            point=outer.globalToLocal(host.localToGlobal(point));
            host=outer;host.addChildAt(layer,host.getChildIndex(field));
         }
         else host.addChild(layer);
         var started:int = 0;
         var requested:int = getTimer();
         var lastFrame:int = -1;
         var animate:Function;
         var cleanup:Function;
         cleanup = function(event:Event = null):void
         {
            if(event != null && event.target != layer) return;
            layer.removeEventListener(Event.ENTER_FRAME,animate);
            layer.removeEventListener(Event.REMOVED_FROM_STAGE,cleanup);
            image.bitmapData = null;
            if(layer.parent != null) layer.parent.removeChild(layer);
         };
         animate = function(event:Event):void
         {
            if(host.stage == null){cleanup();return;}
            var frames:Array = FRasterTemplateFxCache[clip.atlas] as Array;
            if(frames == null){layer.visible=false;if(getTimer()-requested>5000) cleanup();return;}
            if(started == 0) started=getTimer()+delay;
            var elapsed:int = getTimer()-started;
            if(elapsed >= clip.timeline.length*40){cleanup();return;}
            if(elapsed < 0){layer.visible=false;return;}
            layer.visible=true;
            var frame:int = int(elapsed/40);
            if(frame != lastFrame)
            {
               image.bitmapData = frames[int(clip.timeline[frame])];
               image.width=width;image.height=height;
               image.x=-width*Number(clip.pivot[0]);image.y=-height*Number(clip.pivot[1]);
               if(clip.tint != null) image.transform.colorTransform=new flash.geom.ColorTransform(Number(clip.tint[0]),Number(clip.tint[1]),Number(clip.tint[2]),1);
               lastFrame=frame;
            }
            if(follow && caster.stage != null) point = host.globalToLocal(caster.localToGlobal(new Point(0,0)));
            layer.x=point.x;layer.y=point.y;layer.scaleX=mirrored ? -1 : 1;
            var duration:Number=clip.timeline.length*40;
            var fade:Number=Math.min(80,duration/4);
            layer.alpha=Math.min(1,(elapsed+40)/fade,Math.max(0,(duration-elapsed)/Math.min(120,duration/4)));
         };
         layer.addEventListener(Event.ENTER_FRAME,animate);
         layer.addEventListener(Event.REMOVED_FROM_STAGE,cleanup);
         animate(null);
      }
