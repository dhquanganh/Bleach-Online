      private static function KabutoVfxManifest() : Object
      {
         return __KABUTO_VFX_MANIFEST__;
      }

      private static function PreloadKabutoVfx() : void
      {
         var manifest:Object = KabutoVfxManifest();
         for(var name:String in manifest.atlases) LoadKabutoVfxAtlas(name,manifest.atlases[name]);
      }

      private static function LoadKabutoVfxAtlas(name:String, meta:Object) : void
      {
         if(FKabutoFxCache[name] != null || FKabutoFxLoading[name]) return;
         FKabutoFxLoading[name] = true;
         var loader:Loader = new Loader();
         var complete:Function;
         var failed:Function;
         var release:Function = function():void
         {
            loader.contentLoaderInfo.removeEventListener(Event.COMPLETE,complete);
            loader.contentLoaderInfo.removeEventListener(IOErrorEvent.IO_ERROR,failed);
            loader.contentLoaderInfo.removeEventListener(SecurityErrorEvent.SECURITY_ERROR,failed);
            delete FKabutoFxLoading[name];
         };
         failed = function(event:Event):void {release();loader.unload();};
         complete = function(event:Event):void
         {
            release();
            var source:Bitmap = loader.content as Bitmap;
            if(source == null) {loader.unload();return;}
            var frames:Array = [];
            var w:int = int(source.bitmapData.width/int(meta.columns));
            var h:int = int(source.bitmapData.height/int(meta.rows));
            for(var i:int=0;i<int(meta.columns)*int(meta.rows);i++)
            {
               var cell:BitmapData = new BitmapData(w,h,true,0);
               cell.copyPixels(source.bitmapData,new flash.geom.Rectangle((i%int(meta.columns))*w,int(i/int(meta.columns))*h,w,h),new Point());
               frames.push(cell);
            }
            FKabutoFxCache[name] = frames;
            loader.unload();
         };
         loader.contentLoaderInfo.addEventListener(Event.COMPLETE,complete);
         loader.contentLoaderInfo.addEventListener(IOErrorEvent.IO_ERROR,failed);
         loader.contentLoaderInfo.addEventListener(SecurityErrorEvent.SECURITY_ERROR,failed);
         loader.load(new URLRequest(String(meta.url)));
      }

      private function PlayKabutoVfxAsset(name:String, host:DisplayObjectContainer, point:Point, width:Number, height:Number, delay:int = 0, mirrored:Boolean = false, follow:Boolean = false, behind:Boolean = false) : void
      {
         var manifest:Object = KabutoVfxManifest();
         var clip:Object = manifest.clips[name];
         if(clip == null || host == null || host.stage == null) return;
         LoadKabutoVfxAtlas(String(clip.atlas),manifest.atlases[clip.atlas]);
         var caster:TRole = this;
         var layer:Sprite = new Sprite();
         layer.name = "kabutoSkillVfx";layer.mouseEnabled = false;layer.mouseChildren = false;
         var image:Bitmap = new Bitmap();image.smoothing = true;layer.addChild(image);
         if(clip.softEdges)
         {
            // Keep broad translucent energy from exposing the rectangular cell boundary.
            var edgeMask:Sprite = new Sprite();
            var gradient:Matrix = new Matrix();
            gradient.createGradientBox(width,height,0,-width*Number(clip.pivot[0]),-height*Number(clip.pivot[1]));
            edgeMask.graphics.beginGradientFill(GradientType.RADIAL,[0xFFFFFF,0xFFFFFF],[1,0],[175,255],gradient);
            edgeMask.graphics.drawRect(-width*Number(clip.pivot[0]),-height*Number(clip.pivot[1]),width,height);
            edgeMask.graphics.endFill();
            image.cacheAsBitmap=true;edgeMask.cacheAsBitmap=true;layer.addChild(edgeMask);image.mask=edgeMask;
         }
         if(behind && host.parent != null)
         {
            // Character ZBufferSort rewrites every child index in the role container.
            // Put the eruption behind that container so depth sorting cannot cover roles.
            var field:DisplayObject = host;
            var outer:DisplayObjectContainer = host.parent;
            point=outer.globalToLocal(host.localToGlobal(point));
            host=outer;host.addChildAt(layer,host.getChildIndex(field));
         }
         else host.addChild(layer);
         var started:int = getTimer()+delay;
         var lastFrame:int = -1;
         var animate:Function;
         var cleanup:Function;
         cleanup = function(event:Event = null):void
         {
            if(event != null && event.target != layer) return;
            layer.removeEventListener(Event.ENTER_FRAME,animate);
            layer.removeEventListener(Event.REMOVED_FROM_STAGE,cleanup);
            image.bitmapData = null;
            if(layer.parent != null) layer.parent.removeChild(layer);
         };
         animate = function(event:Event):void
         {
            var elapsed:int = getTimer()-started;
            if(host.stage == null || elapsed >= clip.timeline.length*40){cleanup();return;}
            if(elapsed < 0){layer.visible=false;return;}
            var frames:Array = FKabutoFxCache[clip.atlas] as Array;
            if(frames == null){layer.visible=false;return;}
            layer.visible=true;
            var frame:int = int(elapsed/40);
            if(frame != lastFrame)
            {
               image.bitmapData = frames[int(clip.timeline[frame])];
               image.width=width;image.height=height;
               image.x=-width*Number(clip.pivot[0]);image.y=-height*Number(clip.pivot[1]);
               if(clip.tint != null) image.transform.colorTransform=new flash.geom.ColorTransform(Number(clip.tint[0]),Number(clip.tint[1]),Number(clip.tint[2]),1);
               lastFrame=frame;
            }
            if(follow && caster.stage != null) point = host.globalToLocal(caster.localToGlobal(new Point(0,0)));
            layer.x=point.x;layer.y=point.y;layer.scaleX=mirrored ? -1 : 1;
            var duration:Number=clip.timeline.length*40;
            var fade:Number=Math.min(80,duration/4);
            layer.alpha=Math.min(1,(elapsed+40)/fade,Math.max(0,(duration-elapsed)/Math.min(120,duration/4)));
         };
         layer.addEventListener(Event.ENTER_FRAME,animate);
         layer.addEventListener(Event.REMOVED_FROM_STAGE,cleanup);
         animate(null);
      }

      private function UpdateKabutoVfx() : void
      {
         if(FId < 11120910 || FId > 11120914 || FSkillId < 13120931 || FSkillId > 13120942 || (FIndex != TYPE_ACTIVE_ATTACK && FIndex != TYPE_ACTIVE_SKILL)) return;
         if(!FKabutoVfxStarted)
         {
            FKabutoVfxStarted = true;
            this.ShowKabutoVfx(FSkillId,false);
         }
         var hits:Object = __KABUTO_VFX_FRAMES__;
         if(!FKabutoVfxHit && FFrameIndex >= int(hits[FSkillId]))
         {
            FKabutoVfxHit = true;
            this.ShowKabutoVfx(FSkillId,true);
         }
      }

      private function ShowKabutoVfx(skill:int, impact:Boolean) : void
      {
         if(skill >= 13120939){this.ShowKabutoSummonVfx(skill,impact);return;}
         var host:DisplayObjectContainer = parent;
         if(host == null || host.stage == null) return;
         var origin:Point = host.globalToLocal(localToGlobal(new Point(0,0)));
         var targets:Array = [];
         var seen:Object = {};
         if(FTargets != null && FResults != null)
         {
            for(var n:int=0;n<FTargets.length && n<FResults.length;n++)
            {
               if(FTargets[n] == null || FTargets[n] == this || (FResults[n].CMD != TTargetInfo.CMD_ATTACK && FResults[n].CMD != TTargetInfo.CMD_ATTACKEX) || FResults[n].ResultInfo.HurtHp < 0) continue;
               var key:String=FTargets[n].Camp+"_"+FTargets[n].Pos;
               if(seen[key]) continue;
               seen[key]=true;
               targets.push(host.globalToLocal(FTargets[n].localToGlobal(new Point(0,0))));
            }
         }
         var mirrored:Boolean = scaleX < 0;
         if(!impact)
         {
            if(skill == 13120931)
               PlayKabutoVfxAsset("cutin",this.FBattleStage,new Point(625,325),1120,440,0,false,false);
            else if(skill == 13120934 || skill == 13120935)
               PlayKabutoVfxAsset("healing",host,origin,170,205,0,false,true);
            else if(skill == 13120936 || skill == 13120938)
            {
               var summonPoint:Point=new Point(origin.x+(mirrored ? -65 : 65),origin.y);
               if(FTargets != null)for each(var summonRole:TRole in FTargets)
               {
                  if(summonRole != null && summonRole != this && summonRole.FId >= 11120911 && summonRole.FId <= 11120914)
                  {
                     summonPoint=host.globalToLocal(summonRole.localToGlobal(new Point(0,0)));
                     break;
                  }
               }
               PlayKabutoVfxAsset("summon",host,summonPoint,160,230,0,mirrored,false,true);
            }
            else if(skill == 13120933)
               PlayKabutoVfxAsset("dragon",host,new Point(origin.x+(mirrored ? -65 : 65),origin.y-40),230,190,120,mirrored);
            return;
         }
         if(targets.length == 0) return;
         if(skill == 13120931)
         {
            var center:Point = new Point();
            var minX:Number=100000;var maxX:Number=-100000;var minY:Number=100000;var maxY:Number=-100000;
            for each(var target:Point in targets)
            {
               minX=Math.min(minX,target.x);maxX=Math.max(maxX,target.x);
               minY=Math.min(minY,target.y);maxY=Math.max(maxY,target.y);
            }
            center.x=(minX+maxX)/2;center.y=(minY+maxY)/2;
            var areaWidth:Number=Math.min(650,Math.max(410,maxX-minX+270));
            // One coordinated eruption around the enemy formation, as in the reference.
            PlayKabutoVfxAsset("mystery-flash",host,new Point(center.x,center.y-60),240,230);
            PlayKabutoVfxAsset("snakes",host,center,areaWidth,350,60,mirrored,false,true);
            PlayKabutoVfxAsset("poison",host,center,areaWidth+160,420,420,false);
            ShakeJiraiyaBattle(this.FBattleStage,6,320);
         }
         else if(skill == 13120932 || skill == 13120937)
         {
            for each(var point:Point in targets) PlayKabutoVfxAsset("slash",host,new Point(point.x,point.y-62),145,175,0,mirrored);
         }
         else if(skill == 13120933)
         {
            for each(var hit:Point in targets) PlayKabutoVfxAsset("white-flash",host,new Point(hit.x,hit.y-55),250,250,0,mirrored);
            ShakeJiraiyaBattle(this.FBattleStage,4,240);
         }
      }
