      private function RestoreKabutoAfterRevival() : void
      {
         if(FKabutoDeathStop != null) FKabutoDeathStop();
         TweenUtil.removeTween(this,false);
         TweenUtil.removeTween(FBitmap,false);
         FIsDie = false;
         FSkillId = 0;
         FSkilling = false;
         FWaitDie = false;
         this.FWaitNextCommand = false;
         this.FHideFrame = -1;
         this.FHasFrozen = false;
         FIsAntiColorBlackWhite = false;
         FIsRedAntiColor = false;
         FCallBackDict[TYPE_ACTIVE_DIE] = null;
         FTimeDict[TYPE_ACTIVE_DIE] = 0;
         this.alpha = 1;
         this.visible = true;
         this.filters = [];
         FBitmap.alpha = 1;
         FBitmap.visible = true;
         FBitmap.filters = [];
         FShaderBmp.visible = true;
         FHpBar.visible = true;
         FBuffSprite.visible = true;
         this.FFrozenBitmap.visible = false;
         this.FGiddyBitmap.visible = false;
         FKabutoVfxStarted = false;
         FKabutoVfxHit = false;
         if(FIdlePosition != null)
         {
            this.x = FIdlePosition[0];
            this.y = FIdlePosition[1];
         }
         PlayFightIdle();
         // Restore an actual surface now, before the next native animation tick.
         var kabutoIdleFrame:Object = FCurSequence.GetAnimationFrameByIndex(0);
         if(kabutoIdleFrame != null)
         {
            FBitmap.bitmapData = kabutoIdleFrame.Surface;
            FBitmap.x = -kabutoIdleFrame.Pivot.X;
            FBitmap.y = -kabutoIdleFrame.Pivot.Y;
         }
      }
