      private function UpdateKabutoVfx() : void
      {
         if(FId < 11120910 || FId > 11120914 || FSkillId < 13120931 || FSkillId > 13120942 || (FIndex != TYPE_ACTIVE_ATTACK && FIndex != TYPE_ACTIVE_SKILL)) return;
         if(!FKabutoVfxStarted)
         {
            FKabutoVfxStarted = true;
            this.ShowKabutoVfx(FSkillId,false);
         }
         var hits:Object = __KABUTO_VFX_FRAMES__;
         if(!FKabutoVfxHit && FFrameIndex >= int(hits[FSkillId]))
         {
            FKabutoVfxHit = true;
            this.ShowKabutoVfx(FSkillId,true);
         }
      }

      private function ShowKabutoVfx(skill:int, impact:Boolean) : void
      {
         var host:DisplayObjectContainer = this.FBattleStage;
         if(host == null || host.stage == null) return;
         var caster:TRole = this;
         var origin:Point = host.globalToLocal(localToGlobal(new Point(0,-50)));
         var targets:Array = [];
         var seen:Object = {};
         var key:String;
         var n:int;
         if(FTargets != null && FResults != null)
         {
            for(n=0;n<FTargets.length && n<FResults.length;n++)
            {
               if(FTargets[n] == null || FTargets[n] == this || (FResults[n].CMD != TTargetInfo.CMD_ATTACK && FResults[n].CMD != TTargetInfo.CMD_ATTACKEX) || FResults[n].ResultInfo.HurtHp < 0) continue;
               key = FTargets[n].Camp+"_"+FTargets[n].Pos;
               if(seen[key]) continue;
               seen[key] = true;
               targets.push(host.globalToLocal(FTargets[n].localToGlobal(new Point(0,-58))));
            }
         }
         if(targets.length == 0) targets.push(origin.clone());
         var color:uint = skill == 13120932 ? 0x71D9FF : (skill == 13120934 || skill == 13120935 ? 0x86FFBE : (skill == 13120939 || skill == 13120941 ? 0xFFAC52 : (skill == 13120940 ? 0x96E9FF : 0xD8A2FF)));
         var layer:Sprite = new Sprite();layer.name = "kabutoSkillVfx";layer.mouseEnabled = false;layer.mouseChildren = false;
         layer.filters = [new GlowFilter(color,0.5,6,6,1,1)];host.addChild(layer);
         var started:int = getTimer();
         var lifetime:int = impact ? 360 : 240;
         if(impact && (skill == 13120931 || skill == 13120933 || skill == 13120939)) ShakeJiraiyaBattle(host,skill == 13120931 ? 6 : 4,300);
         var animate:Function;
         var cleanup:Function;
         cleanup = function(event:Event = null):void
         {
            layer.removeEventListener(Event.ENTER_FRAME,animate);
            layer.removeEventListener(Event.REMOVED_FROM_STAGE,cleanup);
            if(layer.parent != null) layer.parent.removeChild(layer);
         };
         animate = function(event:Event):void
         {
            var elapsed:Number = getTimer()-started;
            var t:Number = elapsed/lifetime;
            if(t>=1 || host.stage == null){cleanup();return;}
            if(!impact && caster.stage != null) origin = host.globalToLocal(caster.localToGlobal(new Point(0,-50)));
            layer.alpha = Math.min(1,t*10)*(1-t);
            var g:Graphics = layer.graphics;g.clear();
            var i:int;var j:int;var a:Number;var r:Number;var p:Point;
            if(!impact || skill == 13120934 || skill == 13120935 || skill == 13120936 || skill == 13120938 || skill == 13120942)
            {
               if(skill == 13120936 || skill == 13120938)
               {
                  // Coffin rises from the ground before the dormant Edo role is revealed.
                  var rise:Number = Math.min(1,t*3);
                  g.lineStyle(2,0x332424,1);g.beginFill(0x765647,1);
                  g.drawRect(origin.x+30,origin.y+50-110*rise,42,110*rise);g.endFill();
                  for(j=0;j<4;j++){g.lineStyle(1,0x382B29,0.8);g.moveTo(origin.x+34+j*10,origin.y+50);g.lineTo(origin.x+34+j*10,origin.y+50-108*rise);}
                  for(j=0;j<7;j++){g.lineStyle();g.beginFill(0xEEE9E5,.75*(1-t));g.drawCircle(origin.x+18+j*10,origin.y+45-22*t,8+13*t);g.endFill();}
               }
               g.lineStyle(2,color,0.85);g.drawEllipse(origin.x-46-22*t,origin.y+40,92+44*t,22);
               for(i=0;i<14;i++)
               {
                  a=i*2.399+elapsed*0.004;
                  g.lineStyle(1.5,color,0.8);g.moveTo(origin.x+Math.cos(a)*35,origin.y+40-((i*11+elapsed*.08)%100));
                  g.lineTo(origin.x+Math.cos(a)*35,origin.y+32-((i*11+elapsed*.08)%100));
               }
               return;
            }
            for(i=0;i<targets.length;i++)
            {
               p=targets[i];
               if(skill == 13120931)
               {
                  // White snakes erupt in front of the struck lineup, then dissolve into poison.
                  for(j=0;j<2;j++)
                  {
                     var sx:Number = p.x+(j == 0 ? -25 : 28);
                     var sy:Number = p.y+45;
                     var height:Number = (100+j*28)*Math.min(1,t*6)*(1-Math.max(0,(t-.55)*2));

                     g.lineStyle(18,0xF0ECE9,1);g.moveTo(sx-22,sy);g.curveTo(sx-65,sy-height*.5,sx-6,sy-height*.65);g.curveTo(sx+34,sy-height*.85,sx+12,sy-height);
                     g.lineStyle(1,0x514454,1);g.beginFill(0xF8F6F2,1);g.drawEllipse(sx+3,sy-height-10,28,16);g.endFill();
                     g.beginFill(0xD54364,1);g.drawCircle(sx+23,sy-height-4,2);g.endFill();
                  }
                  // The poison rolls across the floor after the white-snake burst.
                  for(j=0;j<7;j++)
                  {
                     g.lineStyle();g.beginFill(j%2 == 0 ? 0x8632B2 : 0xB75BE0,Math.max(0,t-.2)*.55);
                     g.drawEllipse(p.x-65+j*17+Math.sin(t*9+j)*12,p.y+18-j%3*14,42+35*t,20+24*t);g.endFill();
                  }
               }
               else if(skill == 13120932 || skill == 13120937)
               {
                  for(j=0;j<3;j++)
                  {
                     g.lineStyle(4-j,color,0.9);g.moveTo(p.x-45-j*8,p.y-40+j*15);g.lineTo(p.x+45+j*8,p.y+30-j*10);
                  }
               }
               else if(skill == 13120933)
               {
                  g.lineStyle(2,0xFFF5ED,1);g.beginFill(0xF45362,1);g.drawCircle(p.x,p.y,10+6*Math.sin(t*Math.PI));g.endFill();

               }
               else if(skill == 13120940)
               {
                  for(j=0;j<3;j++)
                  {
                     g.lineStyle(3-j,0xA4F2FF,0.8);g.moveTo(p.x,p.y+j*8);
                     g.curveTo((p.x+origin.x)/2,p.y-60-j*12,origin.x,origin.y+j*8);
                  }
                  g.lineStyle(3,color,0.8);g.drawEllipse(p.x-25,p.y-45,50,95);
               }
               else
               {
                  for(j=0;j<9;j++)
                  {
                     a=j*Math.PI*2/9+t*2;r=12+48*t;
                     g.lineStyle(2,0xFFC677,0.9);g.beginFill(j%2 == 0 ? 0xFFCE65 : 0xEF7042,0.75);
                     g.drawCircle(p.x+Math.cos(a)*r,p.y+Math.sin(a)*r,10*(1-t)+3);g.endFill();
                  }
               }
            }
         };
         layer.addEventListener(Event.ENTER_FRAME,animate);layer.addEventListener(Event.REMOVED_FROM_STAGE,cleanup);animate(null);
      }
