      private function ShowGaaraVfx(skill:int) : void
      {
         var host:DisplayObjectContainer = FBattleStage;
         if(host == null || host.stage == null)
         {
            return;
         }
         var offset:Point = GaaraStrikeOffset();
         var origin:Point = host.globalToLocal(localToGlobal(new Point(offset.x + 70,offset.y)));
         var zero:Point = host.globalToLocal(localToGlobal(new Point()));
         var one:Point = host.globalToLocal(localToGlobal(new Point(1,0)));
         var unit:Number = Point.distance(zero,one);
         var data:Object = GaaraVfxManifest();
         var name:String = String(data.moves[skill]);
         var effects:Array = data.actions[name] as Array;
         if(skill == 13121501)
         {
            GaaraBackdrop(host);
         }
         if(effects != null)
         {
            for each(var effect:String in effects)
            {
               PlayGaaraEffect(effect,host,origin,skill == 13121501 ? 1600 : 1050,unit,0);
            }
         }
         if(skill == 13121502 && FTargets != null && FResults != null)
         {
            var i:int = 0;
            while(i < FTargets.length && i < FResults.length)
            {
               if(FTargets[i] != null && FResults[i].CMD == 8)
               {
                  var at:Point = host.globalToLocal(FTargets[i].localToGlobal(new Point()));
                  PlayGaaraEffect("GAR_shazhidun",host,at,300,unit,0);
               }
               i++;
            }
         }
      }

      private function PlayGaaraEffect(effect:String, host:DisplayObjectContainer, at:Point, limit:Number, unit:Number, delay:int) : void
      {
         var data:Object = GaaraVfxManifest();
         var layers:Array = data.effects[effect] as Array;
         if(layers == null)
         {
            return;
         }
         var left:Number = 0;
         var right:Number = 0;
         var top:Number = 0;
         var bottom:Number = 0;
         for each(var key:String in layers)
         {
            var c:Object = data.clips[key];
            var w:Number = c.worldWidth != null ? Number(c.worldWidth) : c.width / c.renderScale;
            var h:Number = c.worldHeight != null ? Number(c.worldHeight) : c.height / c.renderScale;
            left = Math.min(left,-Number(c.pivot[0]) * w);
            right = Math.max(right,(1 - Number(c.pivot[0])) * w);
            top = Math.min(top,-Number(c.pivot[1]) * h);
            bottom = Math.max(bottom,(1 - Number(c.pivot[1])) * h);
         }
         var fit:Number = Math.min(unit,limit * unit / Math.max(1,right - left,bottom - top));
         if(effect == "Yzwoailuo_aoyi")
         {
            fit *= 2;
         }
         for each(key in layers)
         {
            c = data.clips[key];
            PlayGaaraVfxAsset(key,host,at,(c.worldWidth != null ? Number(c.worldWidth) : c.width / c.renderScale) * fit,(c.worldHeight != null ? Number(c.worldHeight) : c.height / c.renderScale) * fit,delay,scaleX < 0,effect != "GAR_shazhidun",false);
         }
      }
